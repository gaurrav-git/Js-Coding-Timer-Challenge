// Given array — no sorting allowed const nums = [23, 45, 67, 89, 12, 90, 44]; function findSecondLargest(arr) { let first = -Infinity, second = -Infinity; for (let n of arr) { if (n > first) { second = first; first = n; } else if (n > second) { second = n; } } return second; } console.log("Second Largest:", findSecondLargest(nums));
